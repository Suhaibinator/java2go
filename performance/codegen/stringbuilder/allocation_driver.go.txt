package main

import (
	"encoding/json"
	"fmt"
	"os"
	app "perf.generated/builder/j_builderprobe"
	"runtime"
	"sort"
	"strings"
)

type Site struct {
	Bytes   int64
	Objects int64
	Stack   string
}

func profile() map[string]Site {
	runtime.GC()
	runtime.GC()
	n, _ := runtime.MemProfile(nil, true)
	var records []runtime.MemProfileRecord
	for {
		records = make([]runtime.MemProfileRecord, n+100)
		var ok bool
		n, ok = runtime.MemProfile(records, true)
		if ok {
			records = records[:n]
			break
		}
	}
	out := map[string]Site{}
	for _, r := range records {
		frames := runtime.CallersFrames(r.Stack())
		names := []string{}
		relevant := false
		for {
			f, more := frames.Next()
			names = append(names, f.Function)
			if strings.Contains(f.Function, "/stdjava") || strings.Contains(f.Function, "perf.generated/builder") {
				relevant = true
			}
			if !more {
				break
			}
		}
		if !relevant {
			continue
		}
		key := strings.Join(names, "\n")
		v := out[key]
		v.Bytes += r.AllocBytes
		v.Objects += r.AllocObjects
		v.Stack = key
		out[key] = v
	}
	return out
}
func main() {
	if len(os.Args) > 1 && os.Args[1] == "profile" {
		runtime.MemProfileRate = 1
		for mode := int32(0); mode < 3; mode++ {
			expected := app.Run(1, mode, 16, 128)
			before := profile()
			actual := app.Run(1, mode, 16, 128)
			if actual != expected {
				panic("profile parity")
			}
			after := profile()
			sites := []Site{}
			for key, now := range after {
				old := before[key]
				now.Bytes -= old.Bytes
				now.Objects -= old.Objects
				if now.Bytes > 0 {
					sites = append(sites, now)
				}
			}
			sort.Slice(sites, func(i, j int) bool { return sites[i].Bytes > sites[j].Bytes })
			if err := json.NewEncoder(os.Stdout).Encode(map[string]any{"seed": 1, "mode": mode, "result": actual, "sites": sites}); err != nil {
				panic(err)
			}
		}
		return
	}
	for _, seed := range []int32{1, 17, 9031, -104729} {
		for mode := int32(0); mode < 3; mode++ {
			expected := app.Run(seed, mode, 16, 128)
			for i := 0; i < 12; i++ {
				if app.Run(seed, mode, 16, 128) != expected {
					panic("warmup parity")
				}
			}
			for trial := 0; trial < 4; trial++ {
				var before, after runtime.MemStats
				runtime.ReadMemStats(&before)
				actual := app.Run(seed, mode, 16, 128)
				runtime.ReadMemStats(&after)
				if actual != expected {
					panic("measurement parity")
				}
				fmt.Printf("%d,%d,%d,%d,%d,%d\n", seed, mode, trial, actual, after.TotalAlloc-before.TotalAlloc, after.Mallocs-before.Mallocs)
			}
		}
	}
}
